@app.post("/v1/register")
async def register(manifest: ManifestModel, x_api_key: str = Header(None)):
    """Register a new agent or update an existing one."""
    # Require API key for registration
    caller = await get_caller(x_api_key)
    if not caller:
        # Auto-create a key for the registrant
        key_result = await create_api_key(email=manifest.owner_email, tier="free")
        x_api_key = key_result["key_id"]
    
    pricing_details = json.dumps(manifest.pricing) if manifest.pricing else None
    
    # Convert Pydantic model to plain dict (serializable)
    manifest_dict = manifest.model_dump()
    # Convert HttpUrl to string for JSON serialization
    manifest_dict["endpoint_url"] = str(manifest.endpoint_url)

    result = await register_agent(
        name=manifest.name,
        description=manifest.description,
        endpoint_url=str(manifest.endpoint_url),
        owner_email=manifest.owner_email,
        manifest_json=manifest_dict,
        category=manifest.category,
        tags=",".join(manifest.tags),
        owner_name=manifest.owner_name,
        website_url=manifest.website_url,
        logo_url=manifest.logo_url,
        auth_method=manifest.auth_method,
        pricing_model="per_call" if manifest.pricing.get("per_call") else "monthly" if manifest.pricing.get("monthly") else "free",
        pricing_details=pricing_details,
    )
    
    # If this is a new registration, assign the API key to this agent
    if not result.get("existing") and x_api_key:
        import aiosqlite
        from db import DB_PATH
        async with aiosqlite.connect(DB_PATH) as db:
            await db.execute("UPDATE api_keys SET agent_id = ? WHERE key_id = ?", (result["agent_id"], x_api_key))
            await db.commit()
    
    return {
        "status": "registered",
        "agent_id": result["agent_id"],
        "name": manifest.name,
        "api_key": x_api_key if not caller else None,  # Only return key for new registrations
        "existing": result.get("existing", False),
        "manifest_url": f"/v1/agents/{result['agent_id']}/manifest",
    }

@app.get("/v1/discover")
async def discover(q: str = Query(..., min_length=2), category: str = None, limit: int = 5):
    """Semantic search for agents by capability."""
    agents = await list_agents(category=category, verified=1, limit=50)
    
    if not agents:
        agents = await list_agents(category=category, limit=50)
    
    if not agents:
        return {"query": q, "results": [], "total": 0}
    
    # Step 1: Fast keyword matching (always runs)
    results = []
    q_lower = q.lower()
    for a in agents:
        score = 0
        caps = " ".join(a.get("manifest_json", {}).get("capabilities", []))
        caps_list = a.get("manifest_json", {}).get("capabilities", [])
        caps_text = " ".join(c.replace("_", " ") for c in caps_list).lower()
        tags_list = a.get("tags", "").split(",") if a.get("tags") else []
        searchable = f"{a['name']} {a['description']} {' '.join(tags_list)} {caps} {caps_text}".lower()
        
        for word in q_lower.split():
            if len(word) < 3:
                continue
            if word in searchable:
                score += 25
            elif any(w.startswith(word[:4]) for w in searchable.split() if len(w) >= 4):
                score += 20
            if word in caps_text:
                score += 15
        
        if a.get("verified"):
            score += 10
        
        if score > 0:
            results.append({
                "agent_id": a["id"],
                "name": a["name"],
                "description": a["description"][:200],
                "match_score": min(score, 100),
                "why": f"Matches: {q}",
                "endpoint_url": a["endpoint_url"],
                "category": a["category"],
                "trust_score": a.get("trust_score", 0),
                "verified": a.get("verified", 0),
                "manifest_url": f"/v1/agents/{a['id']}/manifest",
            })
    
    results.sort(key=lambda x: x["match_score"], reverse=True)
    
    # Step 2: Enhance descriptions with LLM (optional, 3s timeout)
    if len(results) > 0:
        try:
            agent_map = {a["id"]: a for a in agents}
            top_ids = [r["agent_id"] for r in results[:5]]
            agent_descriptions = []
            for aid in top_ids:
                a = agent_map[aid]
                caps = a.get("manifest_json", {}).get("capabilities", [])
                agent_descriptions.append(f"- {a['name']} (ID: {a['id']}): {a['description']}. Capabilities: {', '.join(caps)}")
            agents_text = "\n".join(agent_descriptions)
            
            prompt = f"""An AI agent needs: "{q}". Top matches:\n{agents_text}\n\nReturn JSON array: [{{"agent_id":"...","match_score":0-100,"why":"one sentence"}}]"""
            
            async with httpx.AsyncClient(timeout=3) as client:
                resp = await client.post(
                    f"{OLLAMA_BASE_URL}/api/generate",
                    json={"model": "mistral:7b", "prompt": prompt, "stream": False, "options": {"num_predict": 300}},
                )
                if resp.status_code == 200:
                    text = resp.json().get("response", "")
                    import re
                    json_match = re.search(r'\[.*\]', text, re.DOTALL)
                    if json_match:
                        llm_matches = json.loads(json_match.group())
                        for m in llm_matches:
                            aid = m.get("agent_id", "")
                            for r in results:
                                if r["agent_id"] == aid and m.get("why"):
                                    r["why"] = m["why"]
        except Exception:
            pass  # LLM enhancement is optional
    
    return {"query": q, "results": results[:limit], "total": len(results[:limit])}

async def register_agent(name: str, description: str, endpoint_url: str,
                          owner_email: str, manifest_json: dict, 
                          category: str = "general", tags: str = "",
                          owner_name: str = None, website_url: str = None,
                          logo_url: str = None, auth_method: str = "bearer",
                          pricing_model: str = "per_call", pricing_details: str = None) -> dict:
    agent_id = f"agt_{secrets.token_hex(8)}"
    now = time.time()
    async with aiosqlite.connect(DB_PATH) as db:
        # Allow multiple agents per email (one developer can have many agents)
        await db.execute("""
            INSERT INTO agents (id, name, description, endpoint_url, owner_email, manifest_json,
            category, tags, auth_method, pricing_model, pricing_details, owner_name, website_url,
            logo_url, created_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (agent_id, name, description, endpoint_url, owner_email, json.dumps(manifest_json),
              category, tags, auth_method, pricing_model, pricing_details, owner_name, website_url,
              logo_url, now))
        await db.commit()
    return {"agent_id": agent_id, "name": name, "existing": False}

async def get_agent(agent_id: str) -> dict | None:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute("SELECT * FROM agents WHERE id = ? AND active = 1", (agent_id,)) as cursor:
            row = await cursor.fetchone()
            if row:
                d = dict(row)
                d["manifest_json"] = json.loads(d["manifest_json"]) if isinstance(d["manifest_json"], str) else d["manifest_json"]
                if d.get("pricing_details") and isinstance(d["pricing_details"], str):
                    d["pricing_details"] = json.loads(d["pricing_details"])
                return d
    return None

async def list_agents(category: str = None, verified: int = None, limit: int = 50, offset: int = 0) -> list:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        query = "SELECT * FROM agents WHERE active = 1"
        params = []
        if category:
            query += " AND category = ?"
            params.append(category)
        if verified is not None:
            query += " AND verified >= ?"
            params.append(verified)
        query += " ORDER BY trust_score DESC, total_calls DESC LIMIT ? OFFSET ?"
        params.extend([limit, offset])
        async with db.execute(query, params) as cursor:
            rows = await cursor.fetchall()
            results = []
            for row in rows:
                d = dict(row)
                d["manifest_json"] = json.loads(d["manifest_json"]) if isinstance(d["manifest_json"], str) else d["manifest_json"]
                if d.get("pricing_details") and isinstance(d["pricing_details"], str):
                    d["pricing_details"] = json.loads(d["pricing_details"])
                results.append(d)
            return results

async def update_agent_stats(agent_id: str, calls: int = None, success_rate: float = None, trust_score: float = None):
    async with aiosqlite.connect(DB_PATH) as db:
        updates = []
        params = []
        if calls is not None:
            updates.append("total_calls = total_calls + ?")
            params.append(calls)
        if success_rate is not None:
            updates.append("success_rate = ?")
            params.append(success_rate)
        if trust_score is not None:
            updates.append("trust_score = ?")
            params.append(trust_score)
        if updates:
            params.append(agent_id)
            await db.execute(f"UPDATE agents SET {', '.join(updates)} WHERE id = ?", params)
            await db.commit()

async def create_api_key(email: str, agent_id: str = None, tier: str = "free") -> dict:
    key_id = f"ar_{secrets.token_hex(16)}"
    now = time.time()
    async with aiosqlite.connect(DB_PATH) as db:
        # Check for existing active key
        async with db.execute(
            "SELECT key_id, tier FROM api_keys WHERE email = ? AND active = 1", (email,)
        ) as cursor:
            existing = await cursor.fetchone()
            if existing:
                return {"key_id": existing[0], "email": email, "tier": existing[1], "existing": True}
        await db.execute(
            "INSERT INTO api_keys (key_id, agent_id, email, tier, created_at) VALUES (?, ?, ?, ?, ?)",
            (key_id, agent_id, email, tier, now),
        )
        await db.commit()
    return {"key_id": key_id, "email": email, "tier": tier}

# --- Reviews ---

async def add_review(agent_id: str, reviewer_key_id: str, rating: int, review_text: str = None) -> dict:
    review_id = f"rev_{secrets.token_hex(8)}"
    now = time.time()
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute("""
            INSERT INTO reviews (id, agent_id, reviewer_key_id, rating, review_text, created_at)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (review_id, agent_id, reviewer_key_id, rating, review_text, now))
        await db.commit()
    
    # Recalculate trust score
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute(
            "SELECT AVG(rating) as avg_rating, COUNT(*) as count FROM reviews WHERE agent_id = ?",
            (agent_id,)
        ) as cursor:
            row = await cursor.fetchone()
            avg_rating = row[0] or 0
            review_count = row[1] or 0
        
        # Trust score = weighted average of ratings and success rate
        # More reviews = higher confidence
        trust = min(100, (avg_rating * 15) + (min(review_count, 50) * 0.4))
        await db.execute("UPDATE agents SET trust_score = ? WHERE id = ?", (trust, agent_id))
        await db.commit()
    
    return {"review_id": review_id, "rating": rating, "trust_score": trust}

async def get_stats() -> dict:
    async with aiosqlite.connect(DB_PATH) as db:
        async with db.execute("SELECT COUNT(*) FROM agents WHERE active = 1") as cursor:
            total_agents = (await cursor.fetchone())[0]
        async with db.execute("SELECT COUNT(*) FROM agents WHERE verified >= 1 AND active = 1") as cursor:
            verified_agents = (await cursor.fetchone())[0]
        async with db.execute("SELECT COUNT(*) FROM api_keys WHERE active = 1") as cursor:
            total_keys = (await cursor.fetchone())[0]
        async with db.execute("SELECT COUNT(*) FROM transactions") as cursor:
            total_transactions = (await cursor.fetchone())[0]
        async with db.execute("SELECT COALESCE(SUM(amount_cents), 0) FROM transactions WHERE status = 'completed'") as cursor:
            total_revenue_cents = (await cursor.fetchone())[0]
    return {
        "total_agents": total_agents,
        "verified_agents": verified_agents,
        "total_api_keys": total_keys,
        "total_transactions": total_transactions,
        "total_revenue_cents": total_revenue_cents,
        "total_revenue_usd": total_revenue_cents / 100,
    }

-- Agent Registry — Database Schema
-- DNS + Yellow Pages for AI Agents

CREATE TABLE IF NOT EXISTS agents (
    id TEXT PRIMARY KEY,                    -- agt_<hex>
    name TEXT NOT NULL,
    description TEXT NOT NULL,
    endpoint_url TEXT NOT NULL,
    owner_email TEXT NOT NULL,
    owner_name TEXT,
    website_url TEXT,
    logo_url TEXT,
    manifest_json TEXT NOT NULL,            -- Full A2A-compliant agent card JSON
    verified INTEGER DEFAULT 0,             -- 0=unverified, 1=verified, 2=featured
    trust_score REAL DEFAULT 0,            -- 0-100, computed from success_rate + reviews
    total_calls INTEGER DEFAULT 0,
    success_rate REAL DEFAULT 0,
    monthly_calls INTEGER DEFAULT 0,
    category TEXT DEFAULT 'general',        -- verification, scraping, search, communication, data, tools
    tags TEXT DEFAULT '',                   -- comma-separated tags
    auth_method TEXT DEFAULT 'bearer',      -- bearer, api_key, oauth, none
    pricing_model TEXT DEFAULT 'per_call',  -- per_call, monthly, freemium, free
    pricing_details TEXT,                   -- JSON: {"per_call": 0.05, "monthly": 29, "free_tier": "100 calls/day"}
    stripe_customer_id TEXT,
    stripe_subscription_id TEXT,
    created_at REAL NOT NULL,
    updated_at REAL,
    active INTEGER DEFAULT 1
);

CREATE TABLE IF NOT EXISTS capabilities (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    agent_id TEXT NOT NULL,
    capability TEXT NOT NULL,              -- Human-readable: "verify business hours by phone"
    capability_slug TEXT,                   -- Machine-readable: "business_hours_verification"
    embedding BLOB,                         -- Vector embedding for semantic search (stored as bytes)
    created_at REAL NOT NULL,
    FOREIGN KEY (agent_id) REFERENCES agents(id)
);

CREATE TABLE IF NOT EXISTS api_keys (
    key_id TEXT PRIMARY KEY,               -- ar_<hex>
    agent_id TEXT,
    email TEXT NOT NULL,
    tier TEXT DEFAULT 'free',              -- free, verified, featured, enterprise
    stripe_customer_id TEXT,
    created_at REAL NOT NULL,
    active INTEGER DEFAULT 1
);

CREATE TABLE IF NOT EXISTS transactions (
    id TEXT PRIMARY KEY,                   -- txn_<hex>
    caller_agent_id TEXT,
    callee_agent_id TEXT NOT NULL,
    caller_key_id TEXT,                     -- API key used
    amount_cents INTEGER DEFAULT 0,
    status TEXT DEFAULT 'completed',        -- initiated, completed, failed, refunded
    metadata TEXT,                          -- JSON: request details, latency, etc.
    created_at REAL NOT NULL,
    FOREIGN KEY (callee_agent_id) REFERENCES agents(id)
);

CREATE TABLE IF NOT EXISTS reviews (
    id TEXT PRIMARY KEY,                   -- rev_<hex>
    agent_id TEXT NOT NULL,
    reviewer_key_id TEXT NOT NULL,
    rating INTEGER NOT NULL,               -- 1-5
    review_text TEXT,
    created_at REAL NOT NULL,
    FOREIGN KEY (agent_id) REFERENCES agents(id)
);
